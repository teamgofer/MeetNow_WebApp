/* CSS */

  height: 60px;
  border-radius: 12px;
  background: #E0E5EC;
  box-shadow: 
    3px 3px 6px #A3B1C6,
    -3px -3px 6px #FFFFFF;

  align-items: center;
  justify-content: center;
  color: #6D7587;
  transition: all 0.3s ease;

.icon-box:hover {
  color: #4a90e2;
}

  box-shadow:

  color: #4a90e2;
}

  color: #66bb6a;
}

.icon-box.danger {
  color: #e57373;
}
